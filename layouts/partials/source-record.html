{{- /* Renders one citation record, matching the style scripts/render-bibliography.py writes into /sources/ so the two renderings agree on author order, wording and links. Expects a dict: (dict "ctx" . "id" "source-id" "record" $record) The record is passed in rather than looked up, because data/sources.yaml is a list and building the id -> record map costs a loop the caller has already done. Dates: verified by probe, not assumed. Hugo hands these back as either a string ("2019-10-01", "2014-05") or an int for a bare year (2016), never a time.Time, so printf "%v" renders all of them correctly and reflect.IsTime errors on an interface{}. Do not reintroduce time formatting here. */ -}} {{- $id := .id -}} {{- with .record -}} {{- $meta := slice -}} {{- with .authors -}}{{- $meta = $meta | append (delimit . ", ") -}}{{- end -}} {{- $meta = $meta | append .publisher -}} {{- if .date -}} {{- $meta = $meta | append (printf "%v" .date) -}} {{- else -}} {{- $meta = $meta | append "no date on the document" -}} {{- end -}}

{{ delimit $meta " · " }}

{{- $links := slice -}} {{- with .url -}} {{- $links = $links | append (printf "the source itself" .) -}} {{- else -}} {{- $links = $links | append "no stable URL found yet" -}} {{- end -}} {{- with .archive -}} {{- $links = $links | append (printf "archived copy" .) -}} {{- end -}} {{- with .doi -}} {{- $links = $links | append (printf "doi:%s" .) -}} {{- end -}}

{{ delimit $links " · " | safeHTML }}

{{- if not .url -}}

Status: no URL. Cited, but not yet publishable.

{{- else if not .verified_on -}}

Status: not verified. The link has not been opened and confirmed, so the claim it carries is unproven.

{{- else if .archive -}}

Status: verified {{ printf "%v" .verified_on }}, archived.

{{- else -}}

Status: verified {{ printf "%v" .verified_on }}, not archived yet.

{{- end -}} {{- else -}}

Source id {{ $id }} is not in data/sources.yaml.

{{- end -}}
